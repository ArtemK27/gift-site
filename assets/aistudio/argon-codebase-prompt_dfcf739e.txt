# 🎄 Чужой смартфон в новогоднюю ночь

Интерактивное веб-приложение-подарок в виде смартфона iOS 18: открытка, экран блокировки, рабочий стол, ПодвалГрам, заметки, фото и VIP Сейф. Персонажи: Сану, Чу Ваньнин и Мо Жань, Kai Angel & 9mice.

## Локальный запуск

Нужны Node.js/npm и Git. Команды выполняются из корня репозитория; в Windows при блокировке PowerShell используй `npm.cmd`/`npx.cmd`.

```text
npm ci
npm run dev
npm run build
npm run preview
```

Стек: React 19, TypeScript, Vite 6, Tailwind CSS 3, Framer Motion 12. Фактические зависимости зафиксированы в `package-lock.json`.

## Проверки

```text
npx playwright install chromium
npm run test:e2e
npx playwright test tests/e2e/home-motion-regressions.spec.ts tests/e2e/milestone3-homescreen-apps.spec.ts tests/e2e/tier4-scenarios.spec.ts
```

Последний запуск трёх перечисленных наборов: **26/26 PASS** (2026-11-05), профили 430×932 и 1280×800. Полный набор в этой сессии не запускался. Исторические падения и ограничения перечислены в TEST_READY.md. Сборка проходит с предупреждением о JS chunk >500 kB.

## Публикация

```text
npm run deploy
```

Команда собирает `dist/` и через `gh-pages` публикует его в существующий `origin`, ветку `gh-pages`, с `.nojekyll`, без force push. Нужны права записи в репозиторий. На GitHub Pages должен быть выбран источник `gh-pages` / `/ (root)`. Исходники ветки `main` коммитятся и отправляются отдельно; деплой не публикует незакоммиченные исходные файлы в `main`.

Не запускай прежний PowerShell-скрипт с `git init` внутри `dist` и `git push -f`. Токены не добавлять в исходники, команды, документацию или логи.

## Состояние приёмки

- Свайпы предварительно одобрены Кириллом на ПК; проверка на настоящем телефоне ещё нужна.
- Отдельное расположение музыкального пузырька принято: пользователь подтвердил, что он больше не пропадает.
- Полный квест на опубликованном сайте проверен обычным Playwright Chromium при 430×932; ошибок консоли/страницы и HTTP ≥400 не было.
- MCP Playwright не запускается из-за отсутствующего настроенного Chrome. Просмотр сохранённых PNG заблокирован инструментом: визуальная приёмка не завершена.
- Промокоды в клиенте — демонстрационные строки, действительность сертификатов не подтверждена. ПИН и подсказки намеренно находятся в исходниках: это игровой квест, не защищённое хранилище.

import { motion, AnimatePresence } from 'framer-motion';
import { ScreenType, AppId } from './types';
import { PhoneShell } from './components/PhoneShell';
import { BaitScreen } from './components/screens/BaitScreen';

// iOS App Window Wrapper with iOS 18 spring physics
const AppWindow: React.FC<{
  children: React.ReactNode;
  onDismiss: () => void;
}> = ({ children }) => {
  return (
    <motion.div
      initial={{ scale: 0.88, opacity: 0, y: 24, borderRadius: 36 }}
      animate={{ scale: 1, opacity: 1, y: 0, borderRadius: 0 }}
      exit={{ scale: 0.84, opacity: 0, y: -24, borderRadius: 44 }}
      transition={{
        type: 'spring',
        stiffness: 320,
        damping: 28,
        mass: 0.75,
      }}
      className="absolute inset-0 w-full h-full z-40 bg-[#000000] overflow-hidden shadow-2xl flex flex-col will-change-transform"
    >
      <div className="flex-1 w-full h-full overflow-hidden flex flex-col">
        {children}
      </div>
    </motion.div>
  );
};

const MainAppContent: React.FC = () => {
  const [screen, setScreen] = useState<ScreenType>('bait');
  const [activeApp, setActiveApp] = useState<AppId>(null);
  const { openThread } = useMessenger();

  // Prime Web Audio API on first user interaction
  useEffect(() => {
    const handleFirstTouch = () => {
      primeAudioContext();
      window.removeEventListener('pointerdown', handleFirstTouch);
    };
    window.addEventListener('pointerdown', handleFirstTouch, { once: true });
    return () => window.removeEventListener('pointerdown', handleFirstTouch);
  }, []);

  const handleUnlockStart = () => {
    setScreen('lock');
  };

  const handleUnlockPhone = () => {
    setScreen('home');
  };

  const handleOpenNotification = (targetApp: string) => {
    setScreen('home');
    if (
      targetApp === 'messenger' ||
      targetApp === 'safe' ||
      targetApp === 'notes' ||
      targetApp === 'gallery' ||
      targetApp === 'settings' ||
      targetApp === 'clock' ||
      targetApp === 'weather'
    ) {
      setActiveApp(targetApp as AppId);
    }
  };

  const handleOpenApp = (appId: AppId) => {
    setActiveApp(appId);
  };

  const handleCloseApp = () => {
    if (activeApp === 'messenger') openThread(null);
    setActiveApp(null);
  };

  // If on BaitScreen, render full cozy card experience with responsive chassis
  if (screen === 'bait') {
    return (
      <div className="w-full h-[100dvh] sm:p-6 flex items-center justify-center bg-[#010103] overflow-hidden select-none">
        {/* Outer ambient desktop glow */}
        <div className="hidden sm:block absolute w-[540px] h-[860px] rounded-full bg-gradient-to-tr from-rose-950/20 via-sky-950/15 to-purple-950/20 blur-3xl pointer-events-none" />

        <div className="w-full h-[100dvh] rounded-none border-0 sm:w-[430px] sm:h-[min(932px,calc(100dvh-48px))] sm:rounded-[56px] sm:border-[10px] sm:border-[#232328] sm:ring-1 sm:ring-white/10 sm:shadow-[0_25px_80px_rgba(0,0,0,0.9)] mx-auto relative overflow-hidden bg-[#000000]">
          <BaitScreen onUnlockStart={handleUnlockStart} />
        </div>
      </div>
    );
  }

  // OS Interface (LockScreen & HomeScreen & Apps)
  return (
    <PhoneShell
      activeApp={activeApp}
      onHomePress={handleCloseApp}
      showDynamicIsland={true}
    >
      <div className="relative w-full h-full overflow-hidden">
        <AnimatePresence mode="wait">
          {screen === 'lock' && (
            <motion.div
              key="lock-screen"
              initial={{ opacity: 0, scale: 0.96 }}
              animate={{ opacity: 1, scale: 1 }}
              exit={{ opacity: 0, y: -70 }}
              transition={{
                type: 'spring',
                stiffness: 340,
                damping: 28,
              }}
              className="absolute inset-0 w-full h-full"
            >
              <LockScreen
                onUnlock={handleUnlockPhone}
                onOpenNotification={handleOpenNotification}
              />
            </motion.div>
          )}

          {screen === 'home' && (
            <motion.div
              key="home-screen"
              initial={{ opacity: 0, scale: 0.94 }}
              animate={{ opacity: 1, scale: 1 }}
              transition={{
                type: 'spring',
                stiffness: 340,
                damping: 28,
              }}
              className="absolute inset-0 w-full h-full"
            >
              <HomeScreen onOpenApp={handleOpenApp} />
            </motion.div>
          )}
        </AnimatePresence>

        {/* Floating App Windows with iOS Swipe-Up Gesture */}
        <AnimatePresence>
          {activeApp && (
            <AppWindow key={`app-window-${activeApp}`} onDismiss={handleCloseApp}>
              {activeApp === 'messenger' && (
                <MessengerApp onBack={handleCloseApp} />
              )}
              {activeApp === 'safe' && (
                <SafeApp onBack={handleCloseApp} />
              )}
              {activeApp === 'notes' && (
                <NotesApp onBack={handleCloseApp} />
              )}
              {activeApp === 'gallery' && (
                <GalleryApp onBack={handleCloseApp} />
              )}
              {activeApp === 'settings' && (
                <SettingsApp onBack={handleCloseApp} />
              )}
              {activeApp === 'clock' && (
                <ClockApp onBack={handleCloseApp} />
              )}
              {activeApp === 'weather' && (
                <WeatherApp onBack={handleCloseApp} />
              )}
            </AppWindow>
          )}
        </AnimatePresence>
      </div>
    </PhoneShell>
  );
};

export const App: React.FC = () => {
  return (
    <SettingsProvider>
      <AudioProvider>
        <MessengerProvider>
          <MainAppContent />
        </MessengerProvider>
      </AudioProvider>
    </SettingsProvider>
  );
};

interface ClockAppProps {
  onBack: () => void;

type ClockTab = 'world' | 'alarm' | 'stopwatch' | 'timer';

export const ClockApp: React.FC<ClockAppProps> = ({ onBack }) => {
  const [activeTab, setActiveTab] = useState<ClockTab>('world');

  // Real-time tick every second for world clocks & countdown
  const [currentTime, setCurrentTime] = useState(new Date());

  useEffect(() => {
    const timer = setInterval(() => {
      setCurrentTime(new Date());
    }, 1000);
    return () => clearInterval(timer);
  }, []);

  // Compute countdown to New Year 2027
  // Target: Jan 1, 2027 00:00:00
  const targetYear = 2027;
  const targetDate = new Date(targetYear, 0, 1, 0, 0, 0);
  const diffMs = Math.max(0, targetDate.getTime() - currentTime.getTime());
  const countdownDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
  const countdownHours = Math.floor((diffMs / (1000 * 60 * 60)) % 24);
  const countdownMinutes = Math.floor((diffMs / (1000 * 60)) % 60);
  const countdownSeconds = Math.floor((diffMs / 1000) % 60);

  // Time format helper with timezone offset
  const formatCityTime = (offsetHours: number) => {
    const d = new Date(currentTime.getTime() + offsetHours * 3600000);
    return {
      time: d.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
      hoursDiff: offsetHours >= 0 ? `+${offsetHours}` : `${offsetHours}`,
    };
  };

  // ================= ALARMS STATE =================
  const [alarms, setAlarms] = useState([
    {
      id: 'ny-alarm',
      time: '00:00',
      label: 'Новый Год 2027 ✨ (Бой курантов + Салют)',
      ringtone: 'VIPERR — Heavy Metal New Year',
      enabled: true,
      highlight: true,
    },
    {
      id: 'prep-alarm',
      time: '23:50',
      label: 'Открыть шампанское в пентхаусе Сану 🍷',
      ringtone: 'Звон хрусталя и салют',
      enabled: true,
      highlight: false,
    },
    {
      id: 'dumplings-alarm',
      time: '23:55',
      label: 'Снять хунтуни Чу Ваньнина с огня 🥟',
      ringtone: 'Тяньвэнь Перезвон',
      enabled: false,
      highlight: false,
    },
  ]);

  const toggleAlarm = (id: string) => {
    vibrate(20);
    playTapSound(500, 0.04);
    setAlarms((prev) =>
      prev.map((al) => (al.id === id ? { ...al, enabled: !al.enabled } : al))
    );
  };

  // ================= STOPWATCH STATE =================
  const [stopwatchRunning, setStopwatchRunning] = useState(false);
  const [stopwatchTime, setStopwatchTime] = useState(0);
  const [laps, setLaps] = useState<number[]>([]);
  const stopwatchRef = useRef<number | null>(null);
  const lastTimestampRef = useRef<number>(0);

  useEffect(() => {
    if (stopwatchRunning) {
      lastTimestampRef.current = performance.now();
      const update = (now: number) => {
        const delta = now - lastTimestampRef.current;
        lastTimestampRef.current = now;
        setStopwatchTime((prev) => prev + delta);
        stopwatchRef.current = requestAnimationFrame(update);
      };
      stopwatchRef.current = requestAnimationFrame(update);
    } else if (stopwatchRef.current) {
      cancelAnimationFrame(stopwatchRef.current);
    }
    return () => {
      if (stopwatchRef.current) cancelAnimationFrame(stopwatchRef.current);
    };
  }, [stopwatchRunning]);

  const handleStartStopStopwatch = () => {
    vibrate(25);
    playTapSound(stopwatchRunning ? 380 : 620, 0.05);
    setStopwatchRunning((r) => !r);
  };

  const handleLapOrReset = () => {
    vibrate(20);
    playTapSound(480, 0.04);
    if (stopwatchRunning) {
      setLaps((prev) => [stopwatchTime, ...prev]);
    } else {
      setStopwatchTime(0);
      setLaps([]);
    }
  };

  const formatStopwatch = (ms: number) => {
    const minutes = Math.floor(ms / 60000);
    const seconds = Math.floor((ms % 60000) / 1000);
    const hundredths = Math.floor((ms % 1000) / 10);

import { playTapSound, vibrate } from '../utils/haptics';

export const DynamicIsland: React.FC<{ className?: string }> = ({ className = '' }) => {
  const [expanded, setExpanded] = useState(false);
  const islandRef = useRef<HTMLDivElement>(null);
  const reducedMotion = useReducedMotion();
  const { currentTrack, isPlaying, currentTime, duration, togglePlay, prevTrack, nextTrack, seek } = useAudio();
  const total = duration || currentTrack.duration || 147;
  const format = (seconds: number) => `${Math.floor(seconds / 60)}:${String(Math.floor(seconds % 60)).padStart(2, '0')}`;
  const collapse = () => {
    setExpanded(false);
    requestAnimationFrame(() => document.querySelector<HTMLElement>('[aria-label="Открыть музыкальный плеер"]')?.focus());
  };

  useEffect(() => {
    if (!expanded) return;
    const onKey = (event: KeyboardEvent) => { if (event.key === 'Escape') collapse(); };
    const onOutside = (event: PointerEvent) => {
      if (!islandRef.current?.contains(event.target as Node)) setExpanded(false);
    };
    document.addEventListener('keydown', onKey);
    document.addEventListener('pointerdown', onOutside);
    return () => {
      document.removeEventListener('keydown', onKey);
      document.removeEventListener('pointerdown', onOutside);
    };
  }, [expanded]);

  const tap = (action: () => void) => {
    vibrate(15);
    playTapSound(520, 0.04);
    action();
  };

  return (
    <div data-testid="dynamic-island" className={`relative z-50 shrink-0 h-[31px] w-full select-none ${className}`}>
      <div className="absolute top-1 left-3 right-3 flex justify-center pointer-events-none">

        type = 'petal';
        z = 0.6 + Math.random() * 0.8;

        type = 'bokeh';
        z = 0.2 + Math.random() * 0.4;

        x: Math.random() * width,
        y: Math.random() * height,

          ctx.shadowBlur = 8;

      });

    };

    render();

    return () => {
      window.removeEventListener('resize', handleResize);
      window.removeEventListener('pointermove', handlePointerMove);
      window.removeEventListener('touchmove', handlePointerMove);
      cancelAnimationFrame(animationFrameId);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      className="absolute inset-0 pointer-events-none z-0 w-full h-full"
    />
  );

import { AppId } from '../../types';
import { vibrate, playTapSound } from '../../utils/haptics';
import { PAGE_0_APPS, PAGE_1_APPS } from '../home/homeAppsData';
import { AppIcon } from '../home/AppIcon';
import { Page0Widgets, Page1Widgets } from '../home/HomeWidgets';

interface HomeScreenProps {
  onOpenApp: (app: AppId) => void;
}

export const HomeScreen: React.FC<HomeScreenProps> = ({ onOpenApp }) => {
  const [currentPage, setCurrentPage] = useState<number>(0);
  const [toastMessage, setToastMessage] = useState<string | null>(null);
  const viewportRef = useRef<HTMLDivElement>(null);
  const x = useMotionValue(0);
  const panStartX = useRef(0);
  const reducedMotion = useReducedMotion();

  const settlePage = (page: number) => {
    const width = viewportRef.current?.clientWidth ?? 0;
    return animate(x, -page * width, reducedMotion
      ? { duration: 0 }
      : { type: 'spring', stiffness: 320, damping: 30, mass: 1 });
  };

  const pageRef = useRef(currentPage);
  pageRef.current = currentPage;
  useEffect(() => {
    let previousWidth = viewportRef.current?.clientWidth ?? 0;
    const observer = new ResizeObserver(() => {
      const width = viewportRef.current?.clientWidth ?? 0;
      if (width === previousWidth) return;
      previousWidth = width;
      x.stop();
      x.set(-pageRef.current * width);
    });
    if (viewportRef.current) observer.observe(viewportRef.current);
    return () => observer.disconnect();
  }, [x]);

  const handleDragEnd = (_: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
    const nextPage = currentPage === 0 && (info.offset.x < -45 || info.velocity.x < -250)
      ? 1
      : currentPage === 1 && (info.offset.x > 45 || info.velocity.x > 250) ? 0 : currentPage;
    if (nextPage !== currentPage) {
      setCurrentPage(nextPage);
      vibrate(15);
      playTapSound(480, 0.03);
    }
    settlePage(nextPage);
  };

  const handleLaunch = (appId: AppId | 'console') => {
    if (appId === 'console') {
      vibrate([25, 50, 25]);
      playTapSound(320, 0.04);
      setToastMessage('ROOT PRIVILEGES: GRANTED 🔪');
      setTimeout(() => setToastMessage(null), 2400);
      return;
    }
    vibrate(25);
    playTapSound(520, 0.05);
    onOpenApp(appId);
  };

  const handleDotClick = (pageIndex: number) => {
    if (currentPage !== pageIndex) {
      vibrate(15);
      playTapSound(450, 0.03);
      setCurrentPage(pageIndex);
      settlePage(pageIndex);
    }
  };

  return (
    <div className="relative w-full h-full flex flex-col justify-between pt-8 pb-3 select-none z-10 overflow-hidden font-sans">
      {/* Horizontal Draggable SpringBoard Pages */}
      <div ref={viewportRef} className="relative flex-1 min-h-0 w-full overflow-hidden flex flex-col">
        <motion.div
          data-testid="home-pages"
          style={{ x }}
          onPanStart={(_, info) => {
            x.stop();
            panStartX.current = x.get() - info.offset.x;
          }}
          onPan={(_, info) => {
            const width = viewportRef.current?.clientWidth ?? 0;
            const nextX = panStartX.current + info.offset.x;
            // Follow the finger between pages; resist only at the outside edges.
            x.set(nextX > 0 ? nextX * 0.2 : nextX < -width ? -width + (nextX + width) * 0.2 : nextX);
          }}
          onPanEnd={handleDragEnd}
          className="flex w-full h-full touch-pan-y"
        >
          {/* Page 0: Main / Quest */}
          <div className="w-full h-full shrink-0 flex flex-col justify-between px-6">
            <Page0Widgets onOpenApp={onOpenApp} />
            <div className="grid grid-cols-4 gap-4 py-4 px-1">
              {PAGE_0_APPS.map((app, idx) => (
                <AppIcon
                  key={app.id}
                  name={app.name}
                  icon={app.icon}
                  badge={app.badge}
                  bg={app.bg}
                  iconColor={app.iconColor}
                  delay={idx * 0.04}
                  onClick={() => handleLaunch(app.id)}
                />
              ))}
            </div>
          </div>

          {/* Page 1: Companion Apps */}
          <div className="w-full h-full shrink-0 flex flex-col justify-between px-6">
            <Page1Widgets onOpenApp={onOpenApp} />
            <div className="grid grid-cols-4 gap-4 py-4 px-1">

                <AppIcon
                  key={app.id}
                  name={app.name}
                  icon={app.icon}
                  badge={app.badge}
                  bg={app.bg}
                  iconColor={app.iconColor}
                  delay={idx * 0.04}
                  onClick={() => handleLaunch(app.id)}
                />
              ))}
            </div>
          </div>
        </motion.div>
      </div>

      {/* Synchronized interactive iOS pagination dots */}
      <div className="flex items-center justify-center gap-2 py-2">
        {[0, 1].map((idx) => (
          <button
            key={idx}
            type="button"
            onClick={() => handleDotClick(idx)}
            aria-label={`Страница ${idx + 1}`}
            aria-current={currentPage === idx ? 'page' : undefined}
            className={`rounded-full transition-all duration-200 cursor-pointer ${
              currentPage === idx ? 'w-2 h-2 bg-white shadow-sm' : 'w-1.5 h-1.5 bg-white/35 hover:bg-white/60'
            }`}
          />
        ))}
      </div>

      {/* Easter Egg Toast Notification */}
      <AnimatePresence>
        {toastMessage && (
          <motion.div
            initial={{ opacity: 0, y: -20, scale: 0.95 }}
            animate={{ opacity: 1, y: 0, scale: 1 }}
            exit={{ opacity: 0, y: -20, scale: 0.95 }}

          >
            {toastMessage}
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
};

export interface Track {
  id: string;
  title: string;
  artist: string;
  fandom: string;
  badge: string;
  src: string;
  cover: string;
  duration: number; // Duration in seconds
}

export const DEFAULT_TRACKS: Track[] = [

export interface AudioContextType {
  tracks: Track[];
  currentTrackIndex: number;
  currentTrack: Track;
  isPlaying: boolean;
  currentTime: number;
  duration: number;
  play: () => void;
  pause: () => void;
  togglePlay: () => void;
  nextTrack: () => void;
  prevTrack: () => void;
  seek: (seconds: number) => void;
  selectTrack: (index: number) => void;
}

const AudioContext = createContext<AudioContextType | null>(null);

export const AudioProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [tracks] = useState<Track[]>(DEFAULT_TRACKS);
  const [currentTrackIndex, setCurrentTrackIndex] = useState<number>(0);
  const [isPlaying, setIsPlaying] = useState<boolean>(true);
  const [currentTime, setCurrentTime] = useState<number>(42);
  const [duration, setDuration] = useState<number>(DEFAULT_TRACKS[0].duration);

  const audioRef = useRef<HTMLAudioElement | null>(null);
  const isPlayingRef = useRef<boolean>(isPlaying);
  const wasPlayingBeforeHiddenRef = useRef<boolean>(false);
  const fallbackTimerRef = useRef<number | null>(null);

  isPlayingRef.current = isPlaying;
  const currentTrack = tracks[currentTrackIndex] || tracks[0];

  // Helper to safely play audio without unhandled promise rejection
  const safePlayAudio = useCallback((audio: HTMLAudioElement | null) => {
    if (!audio) return;
    const playPromise = audio.play();
    if (playPromise !== undefined) {
      playPromise.catch((err) => {
        // Autoplay policy or abort, safe to log and allow synthetic timer fallback
        console.warn('Audio playback not allowed or aborted:', err?.message || err);
      });
    }
  }, []);

  // Forward declaration of nextTrack for event listeners
  const nextTrackRef = useRef<() => void>(() => {});

  // Initialize HTML5 Audio element once
  useEffect(() => {
    const audio = new Audio();
    audio.preload = 'metadata';
    audio.src = currentTrack.src;
    audioRef.current = audio;

    const onTimeUpdate = () => {
      if (!isNaN(audio.currentTime)) {
        setCurrentTime(Math.floor(audio.currentTime));
      }
    };

    const onLoadedMetadata = () => {
      if (!isNaN(audio.duration) && audio.duration > 0) {
        setDuration(Math.floor(audio.duration));
      }
    };

    const onEnded = () => {
      nextTrackRef.current();
    };

    const onError = () => {
      console.warn('Audio element error loading:', audio.src, '- falling back to visual synthetic progress');
    };

    audio.addEventListener('timeupdate', onTimeUpdate);
    audio.addEventListener('loadedmetadata', onLoadedMetadata);
    audio.addEventListener('ended', onEnded);
    audio.addEventListener('error', onError);

    if (isPlayingRef.current) {
      safePlayAudio(audio);
    }

    return () => {
      audio.pause();
      audio.removeEventListener('timeupdate', onTimeUpdate);
      audio.removeEventListener('loadedmetadata', onLoadedMetadata);
      audio.removeEventListener('ended', onEnded);
      audio.removeEventListener('error', onError);
      audioRef.current = null;
    };
  }, [safePlayAudio]);

  // Update track source on track change
  useEffect(() => {
    const audio = audioRef.current;
    if (!audio) return;

    audio.src = currentTrack.src;
    audio.currentTime = 0;
    setCurrentTime(0);
    setDuration(currentTrack.duration);

    if (isPlayingRef.current) {
      safePlayAudio(audio);
    }
  }, [currentTrackIndex, currentTrack.src, currentTrack.duration, safePlayAudio]);

  // Fallback timer when real audio element cannot play or is paused by autoplay policy
  useEffect(() => {
    if (!isPlaying) {
      if (fallbackTimerRef.current) {
        clearInterval(fallbackTimerRef.current);
        fallbackTimerRef.current = null;
      }
      return;
    }

    fallbackTimerRef.current = window.setInterval(() => {
      const audio = audioRef.current;
      // If audio is actively playing and reporting time, let timeupdate handle it
      if (audio && !audio.paused && !isNaN(audio.currentTime) && audio.currentTime > 0) {
        return;
      }

      // Otherwise simulate playback progress
      setCurrentTime((prev) => {
        const d = duration || currentTrack.duration;
        if (prev >= d) {
          nextTrackRef.current();
          return 0;
        }
        return prev + 1;
      });
    }, 1000);

    return () => {
      if (fallbackTimerRef.current) {
        clearInterval(fallbackTimerRef.current);
        fallbackTimerRef.current = null;
      }
    };
  }, [isPlaying, duration, currentTrack.duration]);

  // R5: Smart Visibility Auto-Pause
  useEffect(() => {
    const handleVisibilityChange = () => {
      if (document.hidden) {
        if (isPlayingRef.current) {
          wasPlayingBeforeHiddenRef.current = true;
          audioRef.current?.pause();
          setIsPlaying(false);
        } else {
          wasPlayingBeforeHiddenRef.current = false;
        }
      } else {
        if (wasPlayingBeforeHiddenRef.current) {
          wasPlayingBeforeHiddenRef.current = false;
          safePlayAudio(audioRef.current);
          setIsPlaying(true);
        }
      }
    };

    document.addEventListener('visibilitychange', handleVisibilityChange);
    return () => {
      document.removeEventListener('visibilitychange', handleVisibilityChange);
    };
  }, [safePlayAudio]);

  const play = useCallback(() => {
    setIsPlaying(true);
    safePlayAudio(audioRef.current);
  }, [safePlayAudio]);

  const pause = useCallback(() => {
    setIsPlaying(false);
    audioRef.current?.pause();
  }, []);

  const togglePlay = useCallback(() => {
    if (isPlayingRef.current) {
      pause();
    } else {
      play();
    }
  }, [play, pause]);

  const nextTrack = useCallback(() => {
    setCurrentTrackIndex((prev) => (prev + 1) % tracks.length);
  }, [tracks.length]);

  const prevTrack = useCallback(() => {
    if (currentTime > 3 && audioRef.current) {
      audioRef.current.currentTime = 0;
      setCurrentTime(0);
    } else {
      setCurrentTrackIndex((prev) => (prev - 1 + tracks.length) % tracks.length);
    }
  }, [currentTime, tracks.length]);

  nextTrackRef.current = nextTrack;

  const seek = useCallback((seconds: number) => {
    if (audioRef.current && !isNaN(seconds)) {
      audioRef.current.currentTime = seconds;
    }
    setCurrentTime(seconds);
  }, []);

  const selectTrack = useCallback((index: number) => {
    if (index >= 0 && index < tracks.length) {
      setCurrentTrackIndex(index);
    }
  }, [tracks.length]);

  return (
    <AudioContext.Provider
      value={{
        tracks,
        currentTrackIndex,
        currentTrack,
        isPlaying,
        currentTime,
        duration: duration || currentTrack.duration,
        play,
        pause,
        togglePlay,
        nextTrack,
        prevTrack,
        seek,
        selectTrack,
      }}
    >
      {children}
    </AudioContext.Provider>
  );
};

export const useAudio = (): AudioContextType => {
  const context = useContext(AudioContext);
  if (!context) {
    throw new Error('useAudio must be used within an AudioProvider');
  }
  return context;
};

  clueRevealed: boolean;
};

const initialState = (): MessengerState => ({
  messages: Object.fromEntries(THREADS.map((thread) => [thread.id, [...thread.messages]])) as Record<ThreadId, ChatMessage[]>,
  revealed: {}, activeThread: null,
  draft: { committee: '', sangwoo: '', wanning: '', kai: '' },
  typing: null, clueRevealed: false,
});

type MessengerActions = {
  state: MessengerState;
  openThread: (id: ThreadId | null) => void;
  setDraft: (id: ThreadId, value: string) => void;
  reveal: (id: string) => void;
  sendText: (id: ThreadId) => void;
  sendSticker: (id: ThreadId, image: string, caption: string) => void;
};
const MessengerContext = createContext<MessengerActions | null>(null);

export const MessengerProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  // Intentionally memory-only: opening/closing apps retains conversations; reloading the page resets them.
  const [state, setState] = useState(initialState);
  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
  const sequence = useRef(0);
  const stickerStarted = useRef(false);
  useEffect(() => {
    const reset = () => {
      timers.current.forEach(clearTimeout);
      timers.current = [];
      sequence.current = 0;
      stickerStarted.current = false;
      setState(initialState());
    };
    window.addEventListener('ios18_quest_reset', reset);
    return () => {
      window.removeEventListener('ios18_quest_reset', reset);
      timers.current.forEach(clearTimeout);
    };
  }, []);

  const later = (callback: () => void, delay: number) => {
    timers.current.push(setTimeout(callback, delay));
  };
  const addMessage = (id: ThreadId, msg: ChatMessage) => {
    setState((prev) => ({ ...prev, messages: { ...prev.messages, [id]: [...prev.messages[id], msg] } }));
  };
  const reply = (id: ThreadId) => {
    const story = THREADS.find((thread) => thread.id === id)!;
    later(() => setState((prev) => ({ ...prev, typing: id })), 450);
    later(() => {
      setState((prev) => ({ ...prev, typing: null }));
      addMessage(id, { ...story.reply, id: `reply-${++sequence.current}`, time: '23:57' });
      playNotificationSound();
    }, 1450);
  };
  const sendText = (id: ThreadId) => {
    const text = state.draft[id].trim();
    if (!text) return;
    addMessage(id, { id: `mine-${++sequence.current}`, sender: 'Вы', avatar: '', text, time: '23:57', isMe: true });
    setState((prev) => ({ ...prev, draft: { ...prev.draft, [id]: '' } }));
    reply(id);
  };
  const sendSticker = (id: ThreadId, image: string, caption: string) => {
    addMessage(id, { id: `sticker-${++sequence.current}`, sender: 'Вы', avatar: '', text: caption, time: '23:57', isMe: true, isSticker: true, stickerUrl: image });
    if (id !== 'committee') { reply(id); return; }
    // Only the first group sticker triggers the quest clue, including rapid repeat taps.
    if (stickerStarted.current) return;
    stickerStarted.current = true;
    later(() => setState((prev) => ({ ...prev, typing: 'committee' })), 450);
    later(() => {
      addMessage(id, { id: 'reaction-kai', ...STICKER_REACTIONS[0], time: '23:57' });
      playNotificationSound();
    }, 1500);
    later(() => {
      setState((prev) => ({ ...prev, typing: null, clueRevealed: true }));
      addMessage(id, { id: 'reaction-clue', ...STICKER_REACTIONS[1], time: '23:58' });
      vibrate(35);
      playNotificationSound();
    }, 3200);
  };
  return <MessengerContext.Provider value={{ state, openThread: (activeThread) => setState((prev) => ({ ...prev, activeThread })), setDraft: (id, value) => setState((prev) => ({ ...prev, draft: { ...prev.draft, [id]: value } })), reveal: (id) => setState((prev) => ({ ...prev, revealed: { ...prev.revealed, [id]: true } })), sendText, sendSticker }}>{children}</MessengerContext.Provider>;
};

export const useMessenger = () => {
  const value = useContext(MessengerContext);
  if (!value) throw new Error('MessengerProvider is required');
  return value;
};

import { setGlobalSound, setGlobalHaptics } from '../utils/haptics';

export type WallpaperId = 'newyear' | 'sangwoo' | 'sisheng';

export interface SettingsState {
  soundEnabled: boolean;
  hapticsEnabled: boolean;
  wallpaper: WallpaperId;
}

export interface SettingsContextType extends SettingsState {
  setSoundEnabled: (enabled: boolean) => void;
  setHapticsEnabled: (enabled: boolean) => void;
  setWallpaper: (wallpaper: WallpaperId) => void;
  resetQuestProgress: () => void;
}

const STORAGE_KEY = 'ios18_gift_settings';

const DEFAULT_SETTINGS: SettingsState = {
  soundEnabled: true,
  hapticsEnabled: true,
  wallpaper: 'newyear',
};

const getStoredSettings = (): SettingsState => {
  if (typeof window === 'undefined' || !window.localStorage) {
    return DEFAULT_SETTINGS;
  }
  try {
    const raw = window.localStorage.getItem(STORAGE_KEY);
    if (!raw) return DEFAULT_SETTINGS;
    const parsed = JSON.parse(raw);
    return {
      soundEnabled: typeof parsed.soundEnabled === 'boolean' ? parsed.soundEnabled : true,
      hapticsEnabled: typeof parsed.hapticsEnabled === 'boolean' ? parsed.hapticsEnabled : true,
      wallpaper: ['newyear', 'sangwoo', 'sisheng'].includes(parsed.wallpaper)
        ? parsed.wallpaper
        : 'newyear',
    };
  } catch {
    return DEFAULT_SETTINGS;
  }
};

export const SettingsContext = createContext<SettingsContextType | null>(null);

export const SettingsProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [settings, setSettings] = useState<SettingsState>(() => getStoredSettings());

  // Keep haptics.ts module synchronized on mount & updates
  useEffect(() => {
    setGlobalSound(settings.soundEnabled);
    setGlobalHaptics(settings.hapticsEnabled);
    try {
      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
    } catch {
      // Ignore storage write error
    }
  }, [settings]);

  const setSoundEnabled = useCallback((enabled: boolean) => {
    setGlobalSound(enabled);
    setSettings((prev) => {
      const next = { ...prev, soundEnabled: enabled };
      try {
        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      } catch {}
      return next;
    });
  }, []);

  const setHapticsEnabled = useCallback((enabled: boolean) => {
    setGlobalHaptics(enabled);
    setSettings((prev) => {
      const next = { ...prev, hapticsEnabled: enabled };
      try {
        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      } catch {}
      return next;
    });
  }, []);

  const setWallpaper = useCallback((wallpaper: WallpaperId) => {
    setSettings((prev) => {
      const next = { ...prev, wallpaper };
      try {
        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      } catch {}
      return next;
    });
  }, []);

  const resetQuestProgress = useCallback(() => {
    try {
      const keysToRemove: string[] = [];
      for (let i = 0; i < window.localStorage.length; i++) {
        const key = window.localStorage.key(i);
        if (key && key !== STORAGE_KEY) {
          keysToRemove.push(key);
        }
      }
      keysToRemove.forEach((k) => window.localStorage.removeItem(k));
      window.dispatchEvent(new CustomEvent('ios18_quest_reset'));
    } catch {
      // Ignore
    }
  }, []);

  const value: SettingsContextType = {
    ...settings,
    setSoundEnabled,
    setHapticsEnabled,
    setWallpaper,
    resetQuestProgress,
  };

  return <SettingsContext.Provider value={value}>{children}</SettingsContext.Provider>;
};

// Web Audio API Sound Synthesizer for zero-latency, 100% reliable SFX
let audioCtx: AudioContext | null = null;

const getAudioContext = (): AudioContext | null => {
  if (typeof window === 'undefined') return null;
  if (!audioCtx) {
    const AudioContextClass = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
    if (AudioContextClass) {
      audioCtx = new AudioContextClass();
    }
  }
  if (audioCtx && audioCtx.state === 'suspended') {
    audioCtx.resume().catch(() => {});
  }
  return audioCtx;

export const primeAudioContext = () => {
  try {
    const ctx = getAudioContext();
    if (!ctx) return;
    if (ctx.state === 'suspended') {
      ctx.resume().catch(() => {});
    }
    // Play a 1-sample silent buffer synchronously on user gesture
    const buffer = ctx.createBuffer(1, 1, 22050);
    const source = ctx.createBufferSource();
    source.buffer = buffer;
    source.connect(ctx.destination);
    source.start(0);
  } catch {
    // Ignore audio priming error
  }
};

export const playTapSound = (freq = 440, duration = 0.04) => {
  if (!soundEnabled) return;
  try {
    const ctx = getAudioContext();
    if (!ctx) return;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();

    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq, ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(150, ctx.currentTime + duration);

    gain.gain.setValueAtTime(0.08, ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);

    osc.connect(gain);
    gain.connect(ctx.destination);

    osc.start();
    osc.stop(ctx.currentTime + duration);
  } catch {
    // Audio context may not be ready
  }
};

export const playGlitchSound = () => {
  if (!soundEnabled) return;
  try {
    const ctx = getAudioContext();
    if (!ctx) return;
    const bufferSize = ctx.sampleRate * 0.45;
    const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
    const data = buffer.getChannelData(0);

    for (let i = 0; i < bufferSize; i++) {
      data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.15));
    }

    const noise = ctx.createBufferSource();
    noise.buffer = buffer;

    const filter = ctx.createBiquadFilter();
    filter.type = 'bandpass';
    filter.frequency.setValueAtTime(1200, ctx.currentTime);
    filter.Q.setValueAtTime(3, ctx.currentTime);

    const gain = ctx.createGain();
    gain.gain.setValueAtTime(0.22, ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.45);

    noise.connect(filter);
    filter.connect(gain);
    gain.connect(ctx.destination);

    noise.start();
  } catch {
    // Ignore
  }
};

export const playNotificationSound = () => {
  if (!soundEnabled) return;
  try {
    const ctx = getAudioContext();
    if (!ctx) return;
    const now = ctx.currentTime;
    
    // Elegant two-tone Apple-like glass ping
    [
      { freq: 587.33, start: 0, dur: 0.12 }, // D5
      { freq: 880.00, start: 0.08, dur: 0.35 } // A5
    ].forEach(({ freq, start, dur }) => {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();

      osc.type = 'triangle';
      osc.frequency.setValueAtTime(freq, now + start);

      gain.gain.setValueAtTime(0.12, now + start);
      gain.gain.exponentialRampToValueAtTime(0.001, now + start + dur);

      osc.connect(gain);
      gain.connect(ctx.destination);

      osc.start(now + start);
      osc.stop(now + start + dur);
    });
  } catch {
    // Ignore
  }
};
